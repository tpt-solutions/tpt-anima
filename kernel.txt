Inference Scheduler - Foundational Rust Data Structure

use std::collections::{HashMap, VecDeque, BinaryHeap};
use std::cmp::Ordering;
use std::time::{Duration, SystemTime, Instant};
use serde::{Deserialize, Serialize};
use uuid::Uuid;

/// Unique identifier for an inference task
pub type InferenceId = Uuid;

/// Unique identifier for a thought bubble (sandboxed execution context)
pub type BubbleId = Uuid;

/// Unique identifier for an agent
pub type AgentId = [u8; 32];

/// Unique identifier for a memory node in SSE
pub type MemoryId = [u8; 32];

/// Hardware resource types
#[derive(Clone, Debug, Serialize, Deserialize, PartialEq, Eq, Hash)]
pub enum HardwareResource {
    CPU,
    GPU(u32),      // GPU index (for multi-GPU systems)
    NPU,           // Neural Processing Unit
    TPU,           // Tensor Processing Unit
    VRAM(u32),     // GPU memory (per device)
    RAM,           // System memory
    NVMe,          // Fast storage (for paging)
}

/// Priority levels for inference scheduling
#[derive(Clone, Debug, Serialize, Deserialize, PartialEq, Eq)]
pub enum InferencePriority {
    Critical = 0,  // System-critical, cannot be preempted
    High = 1,      // User-facing, real-time
    Normal = 2,    // Standard background tasks
    Low = 3,       // Batch processing, can be deferred
    Deferred = 4,  // Only runs when system is idle
}

impl PartialOrd for InferencePriority {
    fn partial_cmp(&self, other: &Self) -> Option<Ordering> {
        Some(self.cmp(other))
    }
}

impl Ord for InferencePriority {
    fn cmp(&self, other: &Self) -> Ordering {
        (*self as u32).cmp(&(*other as u32))
    }
}

/// KV Cache configuration for transformer models
#[derive(Clone, Debug, Serialize, Deserialize)]
pub struct KVCacheConfig {
    /// Maximum sequence length this cache can handle
    pub max_sequence_length: u32,
    
    /// Number of attention heads
    pub num_heads: u32,
    
    /// Dimension per head
    pub head_dim: u32,
    
    /// Number of layers
    pub num_layers: u32,
    
    /// Data type for cache (f16, bf16, f32, etc.)
    pub dtype: String,
    
    /// Total memory footprint (calculated)
    pub memory_footprint_bytes: u64,
    
    /// Whether to use paged attention (for long sequences)
    pub use_paged_attention: bool,
    
    /// Page size for paged attention (in tokens)
    pub page_size_tokens: u32,
}

impl KVCacheConfig {
    /// Calculate memory footprint
    pub fn calculate_footprint(&mut self) {
        // KV cache = 2 (K and V) * num_layers * num_heads * head_dim * dtype_size * seq_len
        let dtype_size = match self.dtype.as_str() {
            "f16" | "bf16" => 2,
            "f32" => 4,
            "f8" | "i8" => 1,
            _ => 2, // default to f16
        };
        
        self.memory_footprint_bytes = 2 * self.num_layers as u64
            * self.num_heads as u64
            * self.head_dim as u64
            * dtype_size as u64
            * self.max_sequence_length as u64;
    }
}

/// The core Inference structure - a unit of AI computation
#[derive(Clone, Debug, Serialize, Deserialize)]

    /// Agent that owns this inference

    /// Human-readable description (for observability)
    pub description: String,
    
    /// Model identifier (e.g., "llama-3-70b", "gpt-4", "custom-model-v2")

    /// Priority level
    pub priority: InferencePriority,
    
    /// Resource requirements
    pub resource_request: ResourceRequest,
    
    /// KV Cache configuration
    pub kv_cache_config: KVCacheConfig,
    
    /// Circuit breaker configuration
    pub circuit_breaker_config: CircuitBreakerConfig,
    
    /// Thermal constraints
    pub thermal_constraints: ThermalConstraints,
    
    /// Memory IDs this inference needs access to (from SSE)
    pub required_memory_ids: Vec<MemoryId>,
    
    /// Current state
    pub state: InferenceState,
    
    /// Assigned thought bubble (if running)
    pub assigned_bubble: Option<BubbleId>,
    
    /// Assigned hardware resources (if allocated)
    pub allocated_resources: Option<AllocatedResources>,
    
    /// Creation timestamp
    pub created_at: SystemTime,
    
    /// Deadline for completion (if any)
    pub deadline: Option<SystemTime>,
    
    /// Estimated compute cost (in FLOPs)
    pub estimated_flops: u64,
    
    /// Actual compute cost (tracked during execution)
    pub actual_flops: u64,

    pub tokens_generated: u32,
    
    /// Maximum tokens to generate
    pub max_tokens: u32,

    pub preemptible: bool,
    
    /// Checkpoint data for resumption (if preempted)
    pub checkpoint: Option<InferenceCheckpoint>,
}

/// State machine for inference lifecycle
#[derive(Clone, Debug, Serialize, Deserialize, PartialEq)]
pub enum InferenceState {
    /// Waiting in queue for resources
    Queued,
    
    /// Resources allocated, ready to start
    Ready,
    
    /// Currently executing
    Running,
    
    /// Preempted by higher priority task (can resume)
    Preempted,
    
    /// Paused due to thermal throttling
    Throttled,
    
    /// Circuit breaker triggered, rolling back
    CircuitBreaking,
    
    /// Completed successfully
    Completed,
    
    /// Failed (error or circuit break limit exceeded)
    Failed,
    
    /// Cancelled by owner
    Cancelled,
}

/// Wrapper for priority queue ordering
#[derive(Clone, Debug)]
pub struct SchedulableInference {
    pub inference: Inference,
    pub priority_score: u32,
}

impl PartialEq for SchedulableInference {
    fn eq(&self, other: &Self) -> bool {
        self.priority_score == other.priority_score
    }
}

impl Eq for SchedulableInference {}

impl PartialOrd for SchedulableInference {
    fn partial_cmp(&self, other: &Self) -> Option<Ordering> {
        Some(self.cmp(other))
    }
}

impl Ord for SchedulableInference {
    fn cmp(&self, other: &Self) -> Ordering {
        // Higher priority score = higher priority (reverse for BinaryHeap)
        other.priority_score.cmp(&self.priority_score)
    }
}

impl Inference {
    /// Create a new inference
    pub fn new(
        owner_agent_id: AgentId,
        description: String,
        model_id: String,
        priority: InferencePriority,
        resource_request: ResourceRequest,
    ) -> Self {
        Self {
            id: Uuid::new_v4(),
            owner_agent_id,
            description,
            model_id,
            priority,
            resource_request,
            kv_cache_config: KVCacheConfig {
                max_sequence_length: 4096,
                num_heads: 32,
                head_dim: 128,
                num_layers: 32,
                dtype: "f16".to_string(),
                memory_footprint_bytes: 0,
                use_paged_attention: true,
                page_size_tokens: 16,
            },
            circuit_breaker_config: CircuitBreakerConfig::default(),
            thermal_constraints: ThermalConstraints::default(),
            required_memory_ids: Vec::new(),
            state: InferenceState::Queued,
            assigned_bubble: None,
            allocated_resources: None,
            created_at: SystemTime::now(),
            deadline: None,
            estimated_flops: 0,
            actual_flops: 0,
            tokens_generated: 0,
            max_tokens: 2048,
            circuit_break_count: 0,
            preemptible: priority != InferencePriority::Critical,
            checkpoint: None,
        }
    }
    
    /// Calculate priority score for scheduling
    pub fn calculate_priority_score(&self) -> u32 {
        let base_priority = match self.priority {
            InferencePriority::Critical => 1000,
            InferencePriority::High => 500,
            InferencePriority::Normal => 100,
            InferencePriority::Low => 50,
            InferencePriority::Deferred => 10,
        };
        
        // Add urgency factor if deadline exists
        let urgency_factor = if let Some(deadline) = self.deadline {
            let now = SystemTime::now();
            if let Ok(time_left) = deadline.duration_since(now) {
                let seconds_left = time_left.as_secs();
                if seconds_left < 60 {
                    500 // Very urgent
                } else if seconds_left < 300 {
                    200 // Urgent
                } else {
                    0
                }
            } else {
                0 // Deadline already passed
            }
        } else {
            0
        };
        
        base_priority + urgency_factor
    }
}

impl InferenceScheduler {
    /// Create a new scheduler
    pub fn new(compute_budget_flops: u64) -> Self {
        Self {
            pending_queue: BinaryHeap::new(),
            running_inferences: HashMap::new(),
            active_bubbles: HashMap::new(),
            hardware_status: HashMap::new(),
            system_thermal_constraints: ThermalConstraints::default(),
            compute_budget_flops,
            current_compute_usage: 0,
            config: SchedulerConfig::default(),
            last_schedule_time: Instant::now(),
        }
    }
    
    /// Submit a new inference for scheduling
    pub fn submit_inference(&mut self, inference: Inference) {
        let priority_score = inference.calculate_priority_score();
        self.pending_queue.push(SchedulableInference {
            inference,
            priority_score,
        });
    }
    
    /// Schedule the next batch of inferences
    pub fn schedule(&mut self) -> Vec<SchedulingDecision> {
        let mut decisions = Vec::new();
        
        // Check thermal limits
        let thermal_ok = self.check_thermal_limits();
        
        // Try to schedule pending inferences
        while let Some(schedulable) = self.pending_queue.pop() {
            if self.running_inferences.len() >= self.config.max_concurrent_inferences as usize {
                self.pending_queue.push(schedulable);
                break;
            }
            
            if !thermal_ok && schedulable.inference.priority < InferencePriority::High {
                self.pending_queue.push(schedulable);
                continue;
            }
            
            // Try to allocate resources
            if let Some(allocated) = self.allocate_resources(&schedulable.inference) {
                let mut inference = schedulable.inference;
                inference.state = InferenceState::Running;
                inference.allocated_resources = Some(allocated);
                
                // Create thought bubble
                let bubble = self.create_thought_bubble(&inference);
                inference.assigned_bubble = Some(bubble.id);
                
                self.active_bubbles.insert(bubble.id, bubble);
                self.running_inferences.insert(inference.id, inference.clone());
                
                decisions.push(SchedulingDecision {
                    inference_id: inference.id,
                    action: SchedulingAction::StartExecution,
                    bubble_id: Some(bubble.id),
                    reason: "Resources available".to_string(),
                });
            } else {
                // Cannot allocate, check for preemption
                if schedulable.inference.preemptible {
                    if let Some(preempted) = self.try_preemption(&schedulable.inference) {
                        decisions.push(preempted);
                    }
                }
                self.pending_queue.push(schedulable);
                break;
            }
        }
        
        self.last_schedule_time = Instant::now();
        decisions
    }
    
    /// Check if thermal limits are OK
    fn check_thermal_limits(&self) -> bool {
        for (_, status) in &self.hardware_status {
            if status.temperature_celsius > self.system_thermal_constraints.max_temp_celsius {
                return false;
            }
        }
        true
    }
    
    /// Try to allocate resources for an inference
    fn allocate_resources(&self, inference: &Inference) -> Option<AllocatedResources> {
        // Simplified allocation logic
        // In reality, this would check actual hardware availability
        
        let mut gpu_compute = HashMap::new();
        let mut vram_bytes = HashMap::new();
        
        for (gpu_id, requested_compute) in &inference.resource_request.gpu_compute {
            if let Some(status) = self.hardware_status.get(&HardwareResource::GPU(*gpu_id)) {
                let available_compute = 1.0 - status.utilization;
                if available_compute >= *requested_compute {
                    gpu_compute.insert(*gpu_id, *requested_compute);
                    vram_bytes.insert(*gpu_id, inference.resource_request.vram_bytes.get(gpu_id).copied().unwrap_or(0));
                } else {
                    return None; // Not enough resources
                }
            } else {
                return None; // GPU not found
            }
        }
        
        Some(AllocatedResources {
            cpu_cores: inference.resource_request.cpu_cores,
            gpu_compute,
            vram_bytes,
            ram_bytes: inference.resource_request.ram_bytes,
            nvme_bandwidth: inference.resource_request.nvme_bandwidth,
            kv_cache_addresses: HashMap::new(),
        })
    }
    
    /// Create a thought bubble for an inference
    fn create_thought_bubble(&self, inference: &Inference) -> ThoughtBubble {
        ThoughtBubble {
            id: Uuid::new_v4(),
            inference_id: inference.id,
            isolation_key: [0u8; 32], // Would be cryptographically generated
            memory_region: MemoryRegion {
                start_address: 0, // Would be allocated from memory manager
                size_bytes: inference.kv_cache_config.memory_footprint_bytes,
                device: HardwareResource::VRAM(0),
                pageable: true,
                page_faults: 0,
            },
            is_active: true,
            created_at: SystemTime::now(),
            last_activity: SystemTime::now(),
            circuit_breaker_state: CircuitBreakerState::default(),
        }
    }
    
    /// Try to preempt a lower-priority inference
    fn try_preemption(&mut self, high_priority: &Inference) -> Option<SchedulingDecision> {
        // Find lowest priority running inference that can be preempted
        let preemptable = self.running_inferences.values()
            .filter(|inf| inf.preemptible && inf.priority > high_priority.priority)
            .min_by_key(|inf| inf.calculate_priority_score());
        
        if let Some(to_preempt) = preemptable {
            let preempted_id = to_preempt.id;
            
            // Create checkpoint
            let checkpoint = InferenceCheckpoint {
                token_position: to_preempt.tokens_generated,
                kv_cache_state: Vec::new(), // Would serialize actual KV cache
                model_weights_offset: 0,
                checkpoint_time: SystemTime::now(),
                checkpoint_size_bytes: 0,
            };
            
            // Update state
            if let Some(inf) = self.running_inferences.get_mut(&preempted_id) {
                inf.state = InferenceState::Preempted;
                inf.checkpoint = Some(checkpoint);
                
                // Remove thought bubble
                if let Some(bubble_id) = inf.assigned_bubble {
                    self.active_bubbles.remove(&bubble_id);
                }
            }
            
            return Some(SchedulingDecision {
                inference_id: preempted_id,
                action: SchedulingAction::Preempt,
                bubble_id: None,
                reason: format!("Preempted by higher priority inference {}", high_priority.id),
            });
        }
        
        None
    }
}

/// Scheduling decision
#[derive(Clone, Debug)]
pub struct SchedulingDecision {
    pub inference_id: InferenceId,
    pub action: SchedulingAction,
    pub bubble_id: Option<BubbleId>,
    pub reason: String,
}

#[derive(Clone, Debug)]
pub enum SchedulingAction {
    StartExecution,
    Preempt,
    Resume,
    Throttle,
    CircuitBreak,
    Complete,
    Fail,
}

Data Structure Explanation
This Rust implementation defines the Cognitive Kernel's Inference Scheduler, which replaces traditional thread/process scheduling with AI-native "Thought Bubble" management.
Core Components:
Inference: A unit of AI computation (not a thread). Contains:
Resource requirements (CPU, GPU, VRAM, RAM)
KV Cache configuration for transformer models
Circuit breaker settings for safety
Thermal constraints
Priority levels and preemption rules
ThoughtBubble: A sandboxed execution context that isolates each inference. Provides:
Cryptographic isolation (fault containment)
Dedicated memory regions
Circuit breaker state tracking
Pageable memory (can swap to NVMe)
InferenceScheduler: The brain that manages everything:
Priority-based queue (Critical > High > Normal > Low > Deferred)
Resource allocation across heterogeneous hardware (CPU/GPU/NPU/TPU)
Thermal monitoring and throttling
Preemption logic (higher priority can interrupt lower priority)
Checkpointing for resumable inferences
Circuit Breaker: Safety mechanism that monitors:
Entropy levels (detects hallucination)
Repetition loops (detects infinite loops)
Causal integrity (ensures outputs make logical sense)
Automatically rolls back or kills dangerous inferences
Hardware Status: Real-time telemetry for every device:
Temperature, power draw, utilization
Available memory
Throttling state
Key Innovations:
Inference-Based Scheduling: Instead of time-slicing threads, the scheduler allocates compute based on attention weights, KV cache pressure, and token generation deadlines.
Unified Memory Fabric: VRAM, RAM, and NVMe are treated as a single addressable space. The scheduler can page KV cache in/out transparently.
Predictive Context Loading: Pre-fetches likely next-token states and SSE memories before they're needed.
Cognitive Circuit Breakers: Built-in safety that prevents hallucination loops and causal violations at the kernel level.
Thermal Awareness: Dynamically allocates FLOPs based on temperature and power constraints, not just raw availability.
This is the "brainstem" of TPT Anima—it's what makes the entire system feel alive and responsive while remaining safe and efficient.
